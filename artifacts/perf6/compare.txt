"""Deterministic identity/order comparison against the saved baseline implementation."""
import json
import random
import sys
from pathlib import Path
from types import ModuleType

root=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(root/'src'))
from aconsole.contracts.app import PaletteItem
from aconsole.contracts.common import PaletteKind
from aconsole.ui.viewmodels.logic.rank import rank_items

baseline=ModuleType('baseline_rank')
path=Path(__file__).with_name('baseline-rank.txt')
exec(compile(path.read_text(),str(path),'exec'),baseline.__dict__)
rng=random.Random(6006)
words=['Open Project','  OPEN  project ','Switch workspace','other','задача','Straße','STRASSE','']
checked=0
for case in range(150):
    items=tuple(PaletteItem(id=f'id-{rng.randrange(12)}',kind=PaletteKind.COMMAND,title=rng.choice(words),
        subtitle=rng.choice(words),keywords=tuple(rng.choices(words,k=2)),aliases=tuple(rng.choices(words,k=3)),
        score_hint=float(rng.randrange(-2,8)),enabled=bool(rng.randrange(2)),disabled_reason=None,
        argument=None,shortcut=None) for _ in range(50))
    for query in words:
        for cutoff in (0,60,100):
            for limit in (None,0,1,8):
                before=baseline.rank_items(items,query,score_cutoff=cutoff,limit=limit)
                after=rank_items(items,query,score_cutoff=cutoff,limit=limit)
                assert len(before)==len(after) and all(a is b for a,b in zip(before,after,strict=True))
                checked+=1
result={'seed':6006,'queries_compared':checked,'same_item_identity_and_order':True}
Path(__file__).with_name('differential.json').write_text(json.dumps(result,indent=2)+'\n')
print(result)
