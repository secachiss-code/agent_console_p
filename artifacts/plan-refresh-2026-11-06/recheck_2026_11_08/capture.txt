class Preview(AConsoleApp):
    def do_activate(self):
        super().do_activate()
        spawn_task(self.ready())
    def _capture(self):
        pass
    async def ready(self):
        for _ in range(100):
            if self.window is not None:
                break
            await asyncio.sleep(.05)
        else:
            raise RuntimeError('window unavailable')
        self.target=self.window
        snapshot=self.client.store.snapshot
        if sys.argv[1].endswith('setup'):
            self.window.show_setup()
            await self.window.setup_wizard.worker
        elif not sys.argv[1].endswith('approvals'):
            plan=snapshot.plans[0]
            self.window._open_plan(plan.id)
            self.window.plan_page.close_button.grab_focus()
            graph=self.window.plan_page
            assert len(graph.node_buttons)==len(plan.nodes)==83
            assert set(graph.node_buttons)=={node.id for node in plan.nodes}
            assert all(parent in graph.node_buttons for node in plan.nodes for parent in node.depends_on)
        await asyncio.sleep(.5)
        self.attempts=0
        self.capture()
    def capture(self):
        clock=self.target.get_frame_clock()
        def painted(clock):
            clock.disconnect(handle)
            try:
                capture_window(self.target,Path(sys.argv[4]))
            except RuntimeError:
                self.attempts+=1
                if self.attempts<20:
                    self.capture();return
                self.failed=True
            spawn_task(self._finish())
        handle=clock.connect('after-paint',painted)
        self.target.queue_draw()

                    command='podman inspect agent-example'),))
            if method=='app/snapshot':
                from aconsole.contracts.app import PlanView
                actual=msgspec.json.decode(Path(__file__).with_name('plan-view.json').read_bytes(),type=PlanView)
                projected=msgspec.structs.replace(actual,project_id=result.snapshot.active_project)
