class Preview(AConsoleApp):
    def do_activate(self):
        super().do_activate()
        spawn_task(self.ready())
    def _capture(self):
        pass
    async def ready(self):
        for _ in range(100):
            if self.window is not None:
                break
            await asyncio.sleep(.05)
        else:
            raise RuntimeError('window unavailable')
        self.target=self.window
        self.dialog=PreferencesDialog(self.window.preferences_vm)
        self.dialog.open(self.window)
        for _ in range(100):
            if not self.dialog.loading:break
            await asyncio.sleep(.05)
        self.target=self.dialog
        if sys.argv[1]=='unsaved':
            self.dialog.font_entry.set_text('Iosevka, Monospace')
            self.dialog.close()
            self.target=self.dialog._close_prompt
        elif sys.argv[1]=='quota-custom':
            self.dialog.quota_warn.set_value(40)
            self.dialog.quota_critical.set_value(50)
            self.dialog.quota_critical.grab_focus()
        await asyncio.sleep(.5)
        if sys.argv[1]=='quota-custom':
            self.dialog.save_button.grab_focus()
            await asyncio.sleep(.2)
        self.attempts=0
        self.capture()
    def capture(self):
        clock=self.target.get_frame_clock()
        def painted(clock):
            clock.disconnect(handle)
            try:
                capture_window(self.target,Path(sys.argv[4]))
            except RuntimeError:
                self.attempts+=1
                if self.attempts<20:
                    self.capture();return
                self.failed=True
            spawn_task(self._finish())
        handle=clock.connect('after-paint',painted)
        self.target.queue_draw()
