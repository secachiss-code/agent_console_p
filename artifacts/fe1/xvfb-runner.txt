import os
from pathlib import Path
import secrets
import struct
import subprocess
import sys
import tempfile
import time

root = Path(__file__).resolve().parents[2]
with tempfile.TemporaryDirectory(prefix='fe1-xvfb-') as directory:
    temp = Path(directory)
    subprocess.run(['/usr/bin/apt-get', 'download', 'xvfb'], cwd=temp, timeout=90, check=True)
    packages = list(temp.glob('xvfb_*.deb'))
    assert len(packages) == 1
    subprocess.run(['/usr/bin/dpkg-deb', '-x', str(packages[0]), str(temp/'package')], timeout=20, check=True)
    number = next(n for n in range(101,120) if not Path(f'/tmp/.X11-unix/X{n}').exists())
    display = str(number).encode()
    cookie = secrets.token_bytes(16)
    auth = temp/'Xauthority'
    auth.write_bytes(struct.pack('>HH',65535,0)+struct.pack('>H',len(display))+display+struct.pack('>H',18)+b'MIT-MAGIC-COOKIE-1'+struct.pack('>H',16)+cookie)
    auth.chmod(0o600)
    server = subprocess.Popen([str(temp/'package/usr/bin/Xvfb'),f':{number}','-screen','0','1280x800x24','-nolisten','tcp','-auth',str(auth)])
    try:
        for _ in range(100):
            if Path(f'/tmp/.X11-unix/X{number}').exists():
                break
            if server.poll() is not None:
                raise RuntimeError('temporary Xvfb failed to start')
            time.sleep(0.05)
        env = dict(os.environ, DISPLAY=f':{number}',XAUTHORITY=str(auth),GDK_BACKEND='x11',GTK_A11Y='none',PYTHONPATH=str(root/'src'))
        env.pop('WAYLAND_DISPLAY',None)
        env.pop('BROADWAY_DISPLAY',None)
        output=root/'artifacts/fe1/ui'
        output.mkdir(parents=True,exist_ok=True)
        env.update(XDG_STATE_HOME=str(temp/'state'),XDG_CONFIG_HOME=str(temp/'config'),XDG_CACHE_HOME=str(temp/'cache'))
        tests=['tests/gui/front/test_fe_races.py','tests/gui/front/test_review_page.py','tests/gui/front/test_git_pane.py','tests/gui/front/test_git_actions.py','tests/gui/front/test_b4a_plan.py','tests/gui/test_core_client.py']
        subprocess.run([sys.executable,'-m','pytest','-o','addopts=','-q',*tests],env=env,cwd=root,timeout=160,check=True)
        for view in ['review','git','plan-graph']:
            for theme,size,name in [('light','1280x800','light'),('dark','1280x800','dark'),('light','700x800','narrow')]:
                subprocess.run([sys.executable,'tools/ui_snapshot.py','--scenario','demo','--view',view,'--theme',theme,'--size',size,'--out',str(output/(view+'-'+name+'.png'))],env=env,cwd=root,timeout=25,check=True)
    finally:
        server.terminate()
        server.wait(timeout=5)
