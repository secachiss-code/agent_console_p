class Preview(AConsoleApp):
    def do_activate(self):
        super().do_activate()
        spawn_task(self.ready())
    def _capture(self):
        pass
    async def ready(self):
        for _ in range(100):
            if self.window is not None:
                break
            await asyncio.sleep(.05)
        else:
            raise RuntimeError('window unavailable')
        self.target=self.window
        if sys.argv[1] in ('new-tab','opening'):
            self.dialog=NewTabDialog(self.window.tabs,lambda _tab:None,lambda error:print(error))
            self.dialog.present(self.window)
            self.dialog.base_branch.set_text('release/next')
            self.dialog.node_row.set_selected(1)
            if sys.argv[1]=='opening':
                self.dialog.open_button.emit('clicked')
            self.target=self.dialog
        elif sys.argv[1]=='project-remove':
            original=Adw.AlertDialog.present
            def present(dialog,parent):
                self.target=dialog
                self.dialog=dialog
                original(dialog,parent)
            Adw.AlertDialog.present=present
            self.window.project_bar.remove_button.emit('clicked')
        else:
            panel=self.window.approval_panel
            for _ in range(100):
                if panel.cards:break
                await asyncio.sleep(.05)
            panel.cards[0].deny_message.set_text('Используй временный каталог вместо удаления данных')
        await asyncio.sleep(.5)
        self.attempts=0
        self.capture()
    def capture(self):
        clock=self.target.get_frame_clock()
        def painted(clock):
            clock.disconnect(handle)
            try:
                capture_window(self.target,Path(sys.argv[4]))
            except RuntimeError:
                self.attempts+=1
                if self.attempts<20:
                    self.capture();return
                self.failed=True
            spawn_task(self._finish())
        handle=clock.connect('after-paint',painted)
        self.target.queue_draw()

with tempfile.TemporaryDirectory(prefix='ux5-preview-') as directory:

    fake=FakeCore(path,'demo');fake.start()
    output=Path(sys.argv[4]);output.unlink(missing_ok=True)
    def factory():
        client=CoreClient(path)
        original=client.request
        async def request(method,params,result_type):
            if method=='session/list':
                return SessionListResult(sessions=())
            if method=='tab/open' and sys.argv[1]=='opening':
                await asyncio.sleep(10)
            result=await original(method,params,result_type)
            if method=='app/snapshot':
                agents=tuple(msgspec.structs.replace(agent,capabilities=agent.capabilities|{'resume','fork'}) for agent in result.snapshot.agents)
                tabs=() if sys.argv[1]=='project-remove' else result.snapshot.tabs
                return msgspec.structs.replace(result,snapshot=msgspec.structs.replace(result.snapshot,agents=agents,tabs=tabs))
            return result
        client.request=request
        snapshot=SnapshotRequest(view='window',theme=sys.argv[2],width=int(sys.argv[3]),height=800,out=output)
        return Preview(client,snapshot=snapshot)
    try:
        result=run_app(factory,['aconsole-ux5-preview'])
    finally:
        fake.stop()
    if not output.is_file():raise RuntimeError('snapshot missing')
    raise SystemExit(result)
