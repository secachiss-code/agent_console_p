import json
import re
import subprocess
from pathlib import Path
files=['tools/hostrun/actions.py','tools/hostrun/server.py','tests/hostrun/test_server.py','tests/hostrun/test_gate_reviews.py']
changed={}
for name in files:
    diff=subprocess.run(['git','diff','--unified=0','--',name],capture_output=True,text=True,check=True).stdout
    lines=set()
    for start,count in re.findall(r'^@@ .*? \+(\d+)(?:,(\d+))? @@',diff,re.M):
        lines.update(range(int(start),int(start)+int(count or 1)))
    changed[str(Path(name).resolve())]=lines
result=subprocess.run(['/home/sas/Projects/sp_aconsole_prototype/.venv-core/bin/ruff','check','--no-force-exclude','--output-format','json',*files],capture_output=True,text=True,check=False)
issues=json.loads(result.stdout)
new=[issue for issue in issues if issue['location']['row'] in changed.get(issue['filename'],set())]
print(json.dumps({'total_diagnostics_in_normally_excluded_files':len(issues),'changed_line_diagnostics':[{'file':i['filename'],'row':i['location']['row'],'code':i['code'],'message':i['message']} for i in new]},ensure_ascii=False,indent=2))
raise SystemExit(bool(new))
