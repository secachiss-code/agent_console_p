import json,time
import gi
gi.require_version('Atspi','2.0')
from gi.repository import Atspi
out=[]
def visit(obj,depth=0):
    if depth>30:return
    try:
        out.append({'role':obj.get_role_name(),'name':obj.get_name(),'description':obj.get_description()})
        for i in range(obj.get_child_count()):visit(obj.get_child_at_index(i),depth+1)
    except Exception as e:out.append({'error':str(e)})
desktop=Atspi.get_desktop(0)
for i in range(desktop.get_child_count()):
    app=desktop.get_child_at_index(i)
    if app.get_name()=='aconsole-des4-preview':visit(app)
print(json.dumps(out,ensure_ascii=False,indent=2))
