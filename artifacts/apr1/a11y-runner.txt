import os
from pathlib import Path
import secrets
import struct
import subprocess
import sys
import argparse
import tempfile
import time
import io
import tarfile

root = Path(__file__).resolve().parents[2]
parser=argparse.ArgumentParser()
parser.add_argument("--captures-only",action="store_true")
parser.add_argument("--tests-only",action="store_true")
parser.add_argument("--baseline",action="store_true")
options=parser.parse_args()
with tempfile.TemporaryDirectory(prefix='apr1-xvfb-') as directory:
    temp = Path(directory)
    subprocess.run(['/usr/bin/apt-get', 'download', 'xvfb'], cwd=temp, timeout=90, check=True)
    packages = list(temp.glob('xvfb_*.deb'))
    assert len(packages) == 1
    subprocess.run(['/usr/bin/dpkg-deb', '-x', str(packages[0]), str(temp/'package')], timeout=20, check=True)
    number = next(n for n in range(101,120) if not Path(f'/tmp/.X11-unix/X{n}').exists())
    display = str(number).encode()
    cookie = secrets.token_bytes(16)
    auth = temp/'Xauthority'
    auth.write_bytes(struct.pack('>HH',65535,0)+struct.pack('>H',len(display))+display+struct.pack('>H',18)+b'MIT-MAGIC-COOKIE-1'+struct.pack('>H',16)+cookie)
    auth.chmod(0o600)
    server = subprocess.Popen([str(temp/'package/usr/bin/Xvfb'),f':{number}','-screen','0','1280x800x24','-nolisten','tcp','-auth',str(auth)])
    try:
        for _ in range(100):
            if Path(f'/tmp/.X11-unix/X{number}').exists():
                break
            if server.poll() is not None:
                raise RuntimeError('temporary Xvfb failed to start')
            time.sleep(0.05)
        env = dict(os.environ, DISPLAY=f':{number}',XAUTHORITY=str(auth),GDK_BACKEND='x11',GTK_A11Y='test',PYTHONPATH=str(root/'src'))
        env.pop('WAYLAND_DISPLAY',None)
        env.pop('BROADWAY_DISPLAY',None)
        output=root/'artifacts/apr1/ui'
        output.mkdir(parents=True,exist_ok=True)
        env.update(XDG_STATE_HOME=str(temp/'state'),XDG_CONFIG_HOME=str(temp/'config'),XDG_CACHE_HOME=str(temp/'cache'))
        env['GTK_A11Y']='atspi'
        env['GTK_USE_PORTAL']='0'
        env['HOME']=str(temp/'home')
        Path(env['HOME']).mkdir()
        speech=(root/'artifacts/apr1/orca-debug.txt')
        orca_log=(root/'artifacts/apr1/orca-start.txt').open('w')
        orca=subprocess.Popen(['/usr/bin/orca','--debug-file',str(speech),'--user-prefs',str(temp/'orca')],env=env,stdout=orca_log,stderr=subprocess.STDOUT)
        try:
            time.sleep(3)
            subprocess.run([sys.executable,str(root/'artifacts/apr1/a11y-preview.txt'),'ru-approvals','light','700',str(root/'artifacts/apr1/orca-preview.png')],env=env,cwd=root,timeout=90,check=True)
        finally:
            orca.terminate()
            try: orca.wait(timeout=5)
            except subprocess.TimeoutExpired:
                orca.kill();orca.wait(timeout=5)
            orca_log.close()
        subprocess.run([sys.executable,str(root/'artifacts/apr1/check-a11y.txt')],cwd=root,check=True)

    finally:
        server.terminate()
        server.wait(timeout=5)
