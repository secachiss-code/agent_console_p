import json,sys,importlib
from pathlib import Path
for name in sys.argv[1:]: importlib.import_module(name)
rows={}; name=''; size=0
for line in Path('/proc/self/smaps').read_text().splitlines():
 p=line.split()
 if '-' in p[0]: name=' '.join(p[5:]) if len(p)>5 else '[anonymous]'
 if p[0]=='Rss:': rows[name]=rows.get(name,0)+int(p[1])/1024
print(json.dumps(sorted(rows.items(),key=lambda x:x[1],reverse=True)[:25],indent=2))
